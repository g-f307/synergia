\begin{dxiconteudo}{ARQUITETURA E LIMITES COMUNS}

  {\dximont\bfseries\fontsize{16}{21}\selectfont\color{hanaroverde}Controles}

  \vspace{0.3em}
  \begin{dxilista}[hanaroverde]

    \item Validação de schema, evidências e abstenção.
    \item Ferramentas somente de leitura e escopo autorizado.

\begin{dxiconteudo}{POC 1 · DIAGNÓSTICO DE QUALIDADE}
\vspace{-0.55cm}
\dxicol{12.8}{%
  {\dximont\bfseries\color{lgred}D.E.E.P AI}\par
  \begin{dxilista}[dxired]
    \item \textbf{Define:} organizar ocorrências determinísticas e orientar a conferência.
    \item \textbf{Explore:} execução, validações, divergências e proveniência autorizada.
    \item \textbf{Experiment:} 12 cenários reservados em CSV e XLSX, três repetições.
    \item \textbf{Productionize:} somente após superar baseline, metas e revisão humana.
  \end{dxilista}

  \vspace{0.35em}
  {\dxiCorpoP\color{dxinavy!85}24 arquivos reservados, 72 tentativas do baseline
  e 72 do agente. Pico RSS do runtime: 2,165 GiB.}
}%
\dxicol{14.0}{%
  {\dximont\bfseries\color{hanaroverde}Resultados medidos}\par
  \begin{tabular}{@{}lrr@{}}
    \textbf{Métrica} & \textbf{Baseline} & \textbf{Agente}\\[0.2em]
    Schema válido & 100\% & 100\%\\
    Aceitação completa & 100\% & 44,44\%\\
    F1 micro & 100\% & 67,78\%\\
    Consistência & 100\% & 29,17\%\\
    Latência média & 0,043 s & 55,865 s\\
  \end{tabular}

  \vspace{0.6em}
  {\dximont\bfseries\color{dxired}Decisão: não promover.}\par
  Evidências sem fundamento, perguntas incompatíveis com o estado e certeza
  indevida mantiveram o agente abaixo das metas.
}
\end{dxiconteudo}

\begin{dxiconteudo}{POC 2 · INVESTIGAÇÃO OPERACIONAL}
\vspace{-0.55cm}
\dxicol{12.8}{%
  {\dximont\bfseries\color{lgred}D.E.E.P AI}\par
  \begin{dxilista}[dxired]
    \item \textbf{Define:} reunir fatos de execução em um resumo rastreável.
    \item \textbf{Explore:} execução, pendências, Workorders e histórico autorizado.
    \item \textbf{Experiment:} quatro casos sintéticos, baseline e três repetições.
    \item \textbf{Productionize:} integrar apenas após qualidade e escolha segura de ferramentas.
  \end{dxilista}

  \vspace{0.35em}
  {\dxiCorpoP\color{dxinavy!85}Ferramentas somente de leitura, seis tentativas
  por rodada e nenhuma ação operacional. Pico RSS v3: 2.460.499.968 bytes.}
}%
\dxicol{14.0}{%
  {\dximont\bfseries\color{hanaroverde}Resultados v3}\par
  \begin{tabular}{@{}lr@{}}
    \textbf{Métrica do agente} & \textbf{Resultado}\\[0.2em]
    Resposta final aceita & 0/12 · 0\%\\
    F1 dos achados & 0\%\\
    Acurácia de ferramentas & 15,48\%\\
    Latência média & 96,779 s\\
    Latência P95 & 140,486 s\\
  \end{tabular}

  \vspace{0.6em}
  {\dximont\bfseries\color{dxired}Decisão: não promover.}\par
  As 12 tentativas atingiram o limite de chamadas. O agente repetiu consultas,
  aplicou filtros incorretos e não concluiu o resumo.
}
\end{dxiconteudo}
